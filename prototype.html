Some black dots and colour artefacts in the decrypted image are known bugs from division-by-zero and channel clamping. See Section 10 of the paper.
⚙ Version 10 Encryption Formula (JavaScript) 
// ── ENCRYPTION (run once on upload) ──────────────────────────────────
// For each pixel group starting at index i (step of 4: R,G,B,A)

let r   = data[i];     // original red value
if (r === 0) { errors++; continue; }  // guard: avoid divide-by-zero

let mdr = data[i] / (i / data[i] % 255);  // MoDula Red — unique key per pixel

data[i+1] = (data[i] / data[i+1]) * 10;         // green → ratio keyed to red
data[i]   = (i / data[i]) % 255;               // red   → position scramble
data[i+2] = (data[i+2] / mdr) * 0.4;           // blue  → scaled by mdr
data[i+3] = 255 - (mdr * 10);                 // alpha → stores mdr as hidden key

// ── DECRYPTION (run in browser on page load) ──────────────────────────

let dmr  = (255 - data[i+3]) / 10;          // recover mdr from alpha channel
data[i]   = data[i] * dmr;                  // restore red
data[i+2] = data[i+2] * dmr * 2.5;          // restore blue
data[i+1] = data[i] / (data[i+1] * 0.1);   // restore green
data[i+3] = 255;                           // restore full opacity
